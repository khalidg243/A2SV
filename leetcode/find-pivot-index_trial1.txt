class Solution:
    def pivotIndex(self, nums: List[int]) -> int:
        prefix = [1] * len(nums)
        prefix[0] = nums[0]
        for i in range(1, len(nums)):
            prefix[i] = nums[i] + prefix[i-1]

        postfix = [1] * len(nums)
        postfix[-1] = nums[-1]
        for i in range(len(nums)-2,-1,-1):
            postfix[i] = nums[i] + postfix[i+1]
        
        for i in range(len(prefix)):
            if len(prefix) == 1:
                return 0
            elif i == 0:
                left = 0
                right = postfix[i + 1]
            elif i == len(prefix) - 1:
                right = 0
                left = prefix[i-1]
            else:
                right = postfix[i+1]
                left = prefix[i-1]
            
            if left == right:
                return i
        return -1
