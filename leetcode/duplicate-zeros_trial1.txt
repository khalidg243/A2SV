class Solution:
    def duplicateZeros(self, arr: List[int]) -> None:
        """
        Do not return anything, modify arr in-place instead.
        """
        i = 0 
        while i < len(arr):
            if arr[i] == 0:
                for x in range(len(arr) - 2, i-1 , -1):
                    arr[x + 1] = arr [x]
                i += 2
            else:
                i += 1
