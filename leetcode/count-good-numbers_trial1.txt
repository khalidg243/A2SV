class Solution:
    def countGoodNumbers(self, n: int) -> int:
        MOD = 10**9 + 7

        def power(base, exp):
            if exp == 0:
                return 1
            
            half = power(base, exp // 2)
            result = (half * half) % MOD

            if exp % 2 == 1:
                result = (result * base) % MOD

            return result

        even_positions = (n+1) // 2
        odd_positions = n // 2

        return (power(5, even_positions) * power(4, odd_positions)) % MOD
