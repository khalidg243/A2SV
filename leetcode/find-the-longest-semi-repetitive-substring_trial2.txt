class Solution:
    def longestSemiRepetitiveSubstring(self, s: str) -> int:
        cnt = 0
        left = 0
        max_cnt = 0
        if len(s) <= 1:
            return len(s)
        for right in range(1,len(s)):
            if s[right] == s[right - 1]:
                cnt += 1
            while cnt > 1:
                if s[left] == s[left + 1]:
                    cnt -= 1
                left += 1
            max_cnt = max(max_cnt, (right - left + 1))
        return max_cnt
