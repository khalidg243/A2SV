class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        left = 0 
        freq = {}
        max_len = 0

        for right, char in enumerate(s):
            freq[char] = freq.get(char,0) + 1
            while freq[char] > 1 and left < right:
                temp = s[left]
                freq[temp] -= 1
                left += 1
            max_len = max(max_len , (right - left + 1))
        
        return max_len
