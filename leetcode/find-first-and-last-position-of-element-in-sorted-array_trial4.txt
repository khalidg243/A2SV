class Solution:
    def searchRange(self, nums: List[int], target: int) -> List[int]:
        l , r = 0 , len(nums) - 1
        temp = target - 1
        ans = [-1 , -1]
        while l <= r:
            mid = l + ((r-l) // 2)
            if nums[mid] <= temp:
                l = mid + 1
            else:
                r = mid - 1 
        
        if l < len(nums) and nums[l] == target:
            ans[0] = l
        else:
            return ans

        r = len(nums) - 1
        
        while l <= r:
            mid = l + ((r-l) // 2)
            if nums[mid] <= target:
                l = mid + 1
            else:
                r = mid - 1 
        
        ans[1] = r 

        return ans
