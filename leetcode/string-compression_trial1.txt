class Solution:
    def compress(self, chars: List[str]) -> int:
        slow = 0 
        fast = 0
         
        while fast < len(chars):
            temp = chars[fast]
            cnt = 0

            while fast < len(chars) and chars[fast] == temp:
                fast += 1
                cnt += 1

            chars[slow] = temp
            slow += 1

            if cnt > 1:
                for digit in str(cnt):
                    chars[slow] = digit
                    slow += 1

        return slow
