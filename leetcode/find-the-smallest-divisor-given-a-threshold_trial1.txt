class Solution:
    def smallestDivisor(self, nums: List[int], threshold: int) -> int:
        l , r = 1, max(nums)

        ans = r

        def can(div):
            total = 0
            for num in nums:
                total += ceil(num / div)
            return total <= threshold

        while l <= r:
            mid = l + ((r-l) // 2)
            if can(mid) :
                ans = mid
                r = mid - 1
            else:
                l = mid + 1
        return ans
