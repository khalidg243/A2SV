class Solution:
    def myPow(self, x: float, n: int) -> float:
        def power(x,n):
            if n == 0:
                return 1

            half = power(x, n // 2)
            res = half * half

            if n % 2 == 1:
                res *= x
            
            return res
        ans = power(x, abs(n))
        if n < 0:
            return 1 / ans
        else:
            return ans
