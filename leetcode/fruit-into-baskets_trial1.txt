class Solution:
    def totalFruit(self, fruits: List[int]) -> int:
        freq = {}
        longest = 0
        left = 0
        for right , fruit in enumerate(fruits):
            freq[fruit] = freq.get(fruit, 0) + 1
            while len(freq) > 2:
                freq[fruits[left]] -= 1
                if freq[fruits[left]] == 0:
                    del freq[fruits[left]]
                left += 1 
            longest = max(longest , right - left + 1)
        return longest
