class Solution:
    def arrangeCoins(self, n: int) -> int:
        l , r = 1 , n
        ans = 0
        while l <= r:
            mid = l + ((r-l) // 2)
            x = (mid*(mid + 1)) / 2
            if x <= n:
                ans = mid
                l = mid + 1
            else:
                r = mid - 1
        return ans
