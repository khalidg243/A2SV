class Solution:
    def predictTheWinner(self, nums: List[int]) -> bool:
        
        def dfs(l,r):

            if l == r:
                return nums[l]

            left_first = nums[l] - dfs(l+1,r)
            right_first = nums[r] - dfs(l,r-1)

            res = max(left_first, right_first)

            return res
        
        difference = dfs(0,len(nums) - 1)

        if difference >= 0:
            return True 
        else:
            return False
