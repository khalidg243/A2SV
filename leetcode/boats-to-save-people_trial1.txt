class Solution:
    def numRescueBoats(self, people: List[int], limit: int) -> int:
        people.sort()
        cnt = 0 
        left , right = 0 , len(people) - 1
        while left <= right:
            if people[right] + people[left] <= limit:
                right -=1
                left += 1
            else:
                right -= 1
            cnt += 1
        return cnt
