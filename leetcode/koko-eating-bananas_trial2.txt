import math

class Solution:
    def minEatingSpeed(self, piles: List[int], h: int) -> int:
        l , r = 1 , max(piles)

        def canEat(speed):
            hours = 0
            for pile in piles:
                hours += math.ceil(pile / speed)
            return hours <= h
        ans = 0
        while l <= r:
            mid = l + ((r-l) // 2)
            if canEat(mid):
                ans = mid
                r = mid - 1
            else:
                l = mid + 1
        
        return ans
