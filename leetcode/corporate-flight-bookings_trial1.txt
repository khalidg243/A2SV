class Solution:
    def corpFlightBookings(self, bookings: List[List[int]], n: int) -> List[int]:
        arr = [0] * n
        for a , b , c in bookings:
            arr[a-1] += c
            if b < n:
                arr[b] -= c 
        for i in range(1,n):
            arr[i] += arr[i-1]
        
        return arr
