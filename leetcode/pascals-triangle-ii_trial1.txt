class Solution:
    def getRow(self, rowIndex: int) -> List[int]:
        
        def build(n):
            if n == 0:
                return [1]

            prev = build(n-1)
            new_row = [1]

            for i in range(1,len(prev)):
                new_row.append(prev[i] + prev[i-1])
            
            new_row.append(1)

            
            return new_row
        return build(rowIndex)
