class NumMatrix:

    def __init__(self, matrix: List[List[int]]):
        rows , cols = len(matrix) , len(matrix[0])
        self.sumMatrix = [[0] * (cols + 1) for i in range(rows + 1)]

        for r in range(1, rows + 1):
            prefix = 0
            for c in range(1, cols + 1):
                prefix += matrix[r-1][c - 1]
                above = self.sumMatrix[r - 1][c]
                self.sumMatrix[r][c] = above + prefix
        print(self.sumMatrix)

    def sumRegion(self, row1: int, col1: int, row2: int, col2: int) -> int:
        topLeft = self.sumMatrix[row1][col1]
        bottomLeft = self.sumMatrix[row2 + 1][col1]
        topRight = self.sumMatrix[row1][col2 + 1]
        bottomRight = self.sumMatrix[row2 + 1][col2+1]
        return bottomRight + topLeft - bottomLeft - topRight
