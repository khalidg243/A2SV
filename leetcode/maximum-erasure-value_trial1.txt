class Solution:
    def maximumUniqueSubarray(self, nums: List[int]) -> int:
        max_sum = 0 
        freq = {}
        left = 0
        temp = 0

        for right , num in enumerate(nums):
            freq[num] = freq.get(num,0) + 1
            temp += num
            while freq[num] > 1 and left < right:
                temp -= nums[left]
                freq[nums[left]] -= 1
                left += 1
            max_sum = max(max_sum, temp)
        return max_sum
