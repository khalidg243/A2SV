class Solution:
    def countPairs(self, nums: List[int], target: int) -> int:
        left = 0 
        right = len(nums) - 1
        nums.sort()
        cnt = 0
        while left < right:
            if nums[left] + nums[right] >= target:
                right -= 1 
            else:
                cnt += right-left
                left += 1
        return cnt
