class Solution:
    def search(self, nums: List[int], target: int) -> int:
        l1 = l2 = 0 
        r1 , r2 = 0, len(nums) -1
        for i in range(1 ,len(nums)):
            if nums[i] < nums[i-1]:
                r1 = i-1
                l2 = i 
                break
        else:
            r1 = len(nums) - 1
            l2 = r2 = None

        while l1 <= r1:
            mid = l1 + ((r1-l1) // 2)
            if nums[mid] == target:
                return mid
            elif nums[mid] < target:
                l1 = mid + 1
            else:
                r1 = mid - 1

        while l2 and l2 <= r2:
            mid = l2 + ((r2-l2) // 2)
            if nums[mid] == target:
                return mid
            elif nums[mid] < target:
                l2 = mid + 1
            else:
                r2 = mid - 1

        return -1
