class Solution:
    def maxDistance(self, position: List[int], m: int) -> int:
        position.sort()
        
        def canPlace(distance):
            last = position[0]
            cnt = 1
            for pos in position[1:]:
                if pos - last >= distance:
                    cnt += 1
                    last = pos
            
                if cnt == m:
                    return True
            
            return False
        
        l , r = 1 , (position[-1] - position[0])
        ans = 0 
        while l <= r:
            mid = l + ((r-l) // 2)
            if canPlace(mid):
                ans = mid
                l = mid + 1
            else:
                r = mid - 1
        
        return ans
